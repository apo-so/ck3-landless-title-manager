
# root - Character who changed government
# scope:old_government - Character's old government type
on_government_change = {
	effect = {
		christian_clergy_learn_liturgical_language_effect = yes

		# MPO remove nomad title
		if = {
			limit = {
				scope:old_government = government_type:nomad_government
			}
			every_held_title = {
				limit = {
					is_nomad_title = yes
					# Keep the holder's preexisting titles and this creation's result during the temporary government change.
					NOT = {
						AND = {
							root = { has_variable = sltm_travel_active }
							OR = {
								var:sltm_pre_create_holder ?= root
								root.var:sltm_travel_returned_title ?= this
							}
						}
					}
				}
				root = {
					destroy_title = prev
				}
			}
		}
		# MPO confederation
		if = {
			limit = {
				NOT = { has_variable = sltm_travel_active }
				is_confederation_member = yes
				# A house-based confederation is left by the house, not by individual members
				confederation = { is_house_based = no }
				NOR = {
					government_has_flag = government_is_nomadic
					government_has_flag = government_is_tribal
					confederation = { has_variable = confederation_ignore_government }
				}
			}
			confederation = {
				remove_confederation_member = root
			}
		}
		# ACH coronation
		if = {
			limit = {
				NOT = { has_variable = sltm_travel_active }
				coronation_trigger = yes
				NOT = { has_game_rule = coronation_laws_off }
			}
			if = {
				limit = {
					has_variable = crowned_emperor_var
					NOT = { has_realm_law = crowned_emperor }
				}
				add_realm_law = crowned_emperor
			}
			else_if = {
				limit = {
					has_variable = crowned_king_var
					NOT = { has_realm_law = crowned_king }
				}
				add_realm_law = crowned_king
			}
		}
		# TGP Mandala clean modifier
		if = {
			limit = {
				NOT = { has_variable = sltm_travel_active }
				scope:old_government = government_type:mandala_government
			}
			#Remove Aspect modifiers/variables
			if = {
				limit = { has_character_modifier = mandala_increments_of_peace_modifier }
				remove_character_modifier = mandala_increments_of_peace_modifier
			}
			if = {
				limit = { has_character_modifier = mandala_creator_modifier }
				remove_character_modifier = mandala_creator_modifier
			}
			remove_variable = mandala_house_power_accumulated_increments_of_peace
			remove_variable = mandala_house_power_accumulated_creator
			#Remove max piety cap modifiers/variables
			if = {
				limit = { has_character_modifier = budding_divinity_modifier }
				remove_character_modifier = budding_divinity_modifier
			}
			if = {
				limit = { has_character_modifier = budding_deity_modifier }
				remove_character_modifier = budding_deity_modifier
			}
			if = {
				limit = { has_character_modifier = budding_godhood_modifier }
				remove_character_modifier = budding_godhood_modifier
			}
			remove_variable = not_subject_to_succession_trials
			remove_variable = tribute_mission_grace
		}
		if = {
			limit = {
				NOT = { has_variable = sltm_travel_active }
				government_has_flag = government_is_mandala
				is_house_head = yes
			}
			#Set the house aspect to default, please
			house = {
				set_house_aspiration = { type = no_aspect }
			}
		}
		# TGP clean ministry
		if = {
			limit = {
				NOT = { has_variable = sltm_travel_active }
				has_title = title:h_china
				scope:old_government = { government_type_has_flag = government_is_celestial }
				tgp_has_access_to_ministry_trigger = no
			}
			every_vassal = {
				limit = { tgp_has_minister_title = yes }
				every_held_title = {
					limit = { tgp_is_minister_title = yes }
					prev = { destroy_title = prev }
				}
			}
		}
		# PAM align theocracy government - the change re-enters this on_action once, then stops
		if = {
			limit = { NOT = { has_variable = sltm_travel_active } }
			pam_align_theocracy_with_faith_government_effect = yes
			pam_strip_illegal_cardinalate_effect = yes
		}
	}
}
