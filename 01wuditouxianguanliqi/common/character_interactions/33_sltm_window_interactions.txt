sltm_menu_create_interaction = {
 category = interaction_category_diplomacy
 common_interaction = yes
 icon = request_maa_interaction
 auto_accept = yes
 use_diplomatic_range = no
 ignores_pending_interaction_block = yes
 is_available = { is_ai = no }
 interface_priority = 129
 send_name = sltm_menu_open_creation
 desc = sltm_menu_create_interaction_desc
 is_shown = { always = no }
 on_accept = {
  hidden_effect = {
   scope:actor = {
    remove_variable = sltm_window_from_manager
    sltm_clw_open_effect = yes
    set_variable = { name = sltm_create_holder value = scope:recipient }
    set_variable = { name = sltm_clw_nominee value = scope:recipient }
    remove_variable = sltm_create_template
    if = { limit = { exists = scope:recipient.primary_title }
     set_variable = { name = sltm_create_template value = scope:recipient.primary_title }
    }
   }
  }
 }
}
