# ===== 旧变量迁移（仅一次）=====
sltm_lvm_migrate_v26_effect = {
    if = {
        limit = { NOT = { has_variable = sltm_lvm_filter_state_migrated_v26 } }
        # 头衔类型 / 等级
        if = { limit = { has_variable = sltm_lvm_filter_title_type } set_variable = { name = sltm_lvm_tf_type value = var:sltm_lvm_filter_title_type } }
        if = { limit = { has_variable = sltm_lvm_filter_title_tier } set_variable = { name = sltm_lvm_tf_tier value = var:sltm_lvm_filter_title_tier } }
        # 旧 status 同时表示待处理/受阻与 主/兼任：分流到 tf_status / tf_role
        if = {
            limit = { has_variable = sltm_lvm_filter_status }
            if = { limit = { var:sltm_lvm_filter_status = flag:pending } set_variable = { name = sltm_lvm_tf_status value = flag:pending } }
            else_if = { limit = { var:sltm_lvm_filter_status = flag:blocked } set_variable = { name = sltm_lvm_tf_status value = flag:blocked } }
            else_if = { limit = { var:sltm_lvm_filter_status = flag:primary } set_variable = { name = sltm_lvm_tf_role value = flag:primary } }
            else_if = { limit = { var:sltm_lvm_filter_status = flag:concurrent } set_variable = { name = sltm_lvm_tf_role value = flag:concurrent } }
        }
        # 旧 scope（人物关系）迁移到 hf_relation
        if = { limit = { has_variable = sltm_lvm_filter_scope } set_variable = { name = sltm_lvm_hf_relation value = var:sltm_lvm_filter_scope } }
        # 旧头衔排序若用了人物字段则退回 status
        if = {
            limit = { has_variable = sltm_lvm_title_sort }
            if = {
                limit = { OR = { var:sltm_lvm_title_sort = flag:holder_rank var:sltm_lvm_title_sort = flag:holder_age } }
                set_variable = { name = sltm_lvm_title_sort value = flag:status }
            }
        }
        set_variable = { name = sltm_lvm_filter_state_migrated_v26 value = flag:yes }
    }
}
