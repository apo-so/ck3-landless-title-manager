# 无地封臣三态 token 效果（V2.7.1）
#
# token 永远存在，只允许：
#   flag:none = 不参与筛选
#   flag:pos  = 必须具有条件
#   flag:neg  = 必须不具有条件
#
# 旧版本“未设置 → pos → neg → remove_variable”的模型会在下一次筛选
# 触发器读取时产生 unset scope 错误。本文件不再删除运行时 token。

sltm_lvm_hf_rebuild_token_list_effect = {
    clear_variable_list = sltm_lvm_hf_tokens
}

# 三态循环：none → pos → neg → none。
sltm_lvm_hf_token_brave_cycle_effect = {
    if = { limit = { var:sltm_lvm_hf_tok_brave = flag:pos } set_variable = { name = sltm_lvm_hf_tok_brave value = flag:neg } }
    else_if = { limit = { var:sltm_lvm_hf_tok_brave = flag:neg } set_variable = { name = sltm_lvm_hf_tok_brave value = flag:none } }
    else = { set_variable = { name = sltm_lvm_hf_tok_brave value = flag:pos } }
    sltm_lvm_hf_rebuild_token_list_effect = yes
}
