# Called only by the confirmed edit, with A already fixed in sltm_edit_title.
# Result 2 means requested. An expired captured source is a failure, never a fallback to current B.

sltm_apply_coa_draft_effect = {
    set_variable = { name = sltm_edit_result_coa value = 3 }
    if = {
        limit = { var:sltm_edit_coa = 1 }
        scope:sltm_edit_title = { generate_coa = yes }
        set_variable = { name = sltm_edit_result_coa value = 2 }
    }
    else_if = {
        limit = { var:sltm_edit_coa = 2 exists = var:sltm_edit_coa_source }
        scope:sltm_edit_title = { set_coa = root.var:sltm_edit_coa_source }
        set_variable = { name = sltm_edit_result_coa value = 2 }
    }
    else_if = {
        limit = { var:sltm_edit_coa = 3 }
        scope:sltm_edit_title = { reset_coa = yes }
        set_variable = { name = sltm_edit_result_coa value = 2 }
    }
}
