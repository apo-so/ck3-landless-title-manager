   limit = {
    OR = {
     AND = { exists = scope:sltm_travel_original_government NOT = { government_type ?= scope:sltm_travel_original_government } }

       exists = scope:sltm_travel_original_primary

        primary_title ?= scope:sltm_created

# A request reserves native cultural mercenary capacity. The engine creates the company at its annual update.
sltm_merc_request_effect = {
    # I18: creation is closed by user request; keep legacy bodies for save-compatible cancellation.
    if = { limit = { always = no }
    var:sltm_create_holder = { save_scope_as = sltm_applicant }
    save_scope_as = sltm_charter_sponsor
    if = { limit = { scope:sltm_applicant = { has_variable = sltm_merc_sponsor } }
        if = { limit = { scope:sltm_applicant.var:sltm_merc_sponsor ?= this }
            scope:sltm_applicant = { sltm_merc_deliver_effect = yes }
        }
        else = { send_interface_toast = { type = event_toast_effect_bad title = SLTM_MC_OTHER left_icon = scope:sltm_applicant custom_tooltip = SLTM_MC_OTHER_DESC } }
    }
    else = {
        scope:sltm_applicant.culture = { save_scope_as = sltm_charter_culture }
        var:sltm_designee_title ?= {
            if = { limit = { tier = tier_county exists = culture } culture = { save_scope_as = sltm_charter_culture } }
        }
        scope:sltm_applicant = {
            clear_variable_list = sltm_merc_before_titles
            set_variable = { name = sltm_merc_before_count value = 0 }
        }
        every_mercenary_company = {
            mercenary_company_leader ?= {
                every_held_title = {
                    limit = { is_mercenary_company = yes }
                    save_scope_as = sltm_charter_prior
                    scope:sltm_applicant = {
                        if = { limit = { NOT = { is_target_in_variable_list = { name = sltm_merc_before_titles target = scope:sltm_charter_prior } } }
                            add_to_variable_list = { name = sltm_merc_before_titles target = scope:sltm_charter_prior }
                            if = { limit = { scope:sltm_charter_prior = {
                                OR = {
                                    var:sltm_merc_origin ?= scope:sltm_charter_culture
                                    AND = { NOT = { has_variable = sltm_merc_origin } holder.culture ?= scope:sltm_charter_culture }
                                }
                            } } change_variable = { name = sltm_merc_before_count add = 1 } }
                        }
                    }
                }
            }
        }
        # A county slot may already belong to another applicant whose new band is not present yet.
        set_local_variable = { name = sltm_committed_bands value = scope:sltm_applicant.var:sltm_merc_before_count }
        every_in_global_list = {
            variable = sltm_pending_charters
            limit = { var:sltm_merc_origin ?= scope:sltm_charter_culture NOT = { exists = var:sltm_merc_candidate } }
            change_local_variable = { name = sltm_committed_bands add = 1 }
        }
        if = {
            limit = { scope:sltm_charter_culture = { culture_number_of_counties > local_var:sltm_committed_bands } }
            scope:sltm_applicant = {
                set_variable = { name = sltm_merc_sponsor value = scope:sltm_charter_sponsor }
                set_variable = { name = sltm_merc_origin value = scope:sltm_charter_culture }
                set_variable = { name = sltm_merc_edit_requested value = $MODE$ }
                set_variable = { name = sltm_merc_units_reserved value = 6 }
                if = { limit = { scope:sltm_charter_culture = { culture_number_of_counties >= 14 } } set_variable = { name = sltm_merc_units_reserved value = 2 } }
                else_if = { limit = { scope:sltm_charter_culture = { culture_number_of_counties >= 7 } } set_variable = { name = sltm_merc_units_reserved value = 3 } }
            }
            scope:sltm_charter_culture = {
                if = { limit = { NOT = { has_variable = sltm_chartered_units } } set_variable = { name = sltm_chartered_units value = 0 } }
                change_variable = { name = sltm_chartered_units add = scope:sltm_applicant.var:sltm_merc_units_reserved }
                sltm_merc_capacity_effect = yes
            }
            if = { limit = { scope:sltm_charter_culture = { has_variable = sltm_charter_capacity_failed } }
                scope:sltm_charter_culture = {
                    change_variable = { name = sltm_chartered_units subtract = scope:sltm_applicant.var:sltm_merc_units_reserved }
                    sltm_merc_capacity_effect = yes
                }
                scope:sltm_applicant = { sltm_merc_clear_request_effect = yes }
                send_interface_toast = { type = event_toast_effect_bad title = SLTM_MC_UNAVAILABLE left_icon = scope:sltm_applicant custom_tooltip = SLTM_MC_RESTART }
            }
            else = {
                add_to_global_variable_list = { name = sltm_pending_charters target = scope:sltm_applicant }
                set_variable = { name = sltm_last_commission value = scope:sltm_applicant }
                send_interface_toast = { type = event_toast_effect_good title = SLTM_MC_QUEUED left_icon = scope:sltm_applicant custom_tooltip = SLTM_MC_QUEUED_DESC }
            }
        }
        else = {
            scope:sltm_applicant = { clear_variable_list = sltm_merc_before_titles remove_variable = sltm_merc_before_count }
            send_interface_toast = { type = event_toast_effect_bad title = SLTM_MC_FULL left_icon = scope:sltm_applicant custom_tooltip = SLTM_MC_FULL_DESC }
        }
    }
    }
}

# Resolve only requests left by the discontinued charter workflow. Never create a company or charge the sponsor.
sltm_merc_cancel_effect = {
    if = {
        limit = { is_alive = no }
        # Archived dead characters may no longer own variables. Read only surviving title and global records.
        if = {
            limit = { is_target_in_global_variable_list = { name = sltm_pending_charters target = this } }
            save_scope_as = sltm_deceased_applicant
            every_mercenary_company = {
                mercenary_company_leader ?= {
                    every_held_title = {
                        limit = { is_mercenary_company = yes var:sltm_merc_claimant ?= scope:sltm_deceased_applicant }
                        remove_variable = sltm_merc_claimant
                    }
                }
            }
            remove_list_global_variable = { name = sltm_pending_charters target = this }
        }
    }
    else_if = {
        limit = {
            OR = {
                has_variable = sltm_merc_sponsor
                is_target_in_global_variable_list = { name = sltm_pending_charters target = this }
            }
        }
        save_scope_as = sltm_legacy_applicant
        set_variable = { name = sltm_merc_resolution value = 4 }
        if = {
            limit = {
                var:sltm_merc_candidate ?= { is_mercenary_company = yes holder ?= scope:sltm_legacy_applicant }
                var:sltm_merc_candidate_company.mercenary_company_leader ?= this
                sltm_legacy_claim_uncontested_trigger = yes
            }
            var:sltm_merc_candidate = {
                set_variable = { name = sltm_merc_native_company value = scope:sltm_legacy_applicant.var:sltm_merc_candidate_company }
                if = {
                    limit = { exists = scope:sltm_legacy_applicant.var:sltm_merc_origin }
                    set_variable = { name = sltm_merc_origin value = scope:sltm_legacy_applicant.var:sltm_merc_origin }
                }
            }
            var:sltm_merc_sponsor ?= {
                set_variable = { name = sltm_last_chartered_title value = scope:sltm_legacy_applicant.var:sltm_merc_candidate }
            }
            set_variable = { name = sltm_merc_resolution value = 3 }
        }
        else_if = {
            limit = {
                exists = var:sltm_merc_previous_captain
                var:sltm_merc_candidate ?= {
                    is_mercenary_company = yes
                    holder ?= scope:sltm_legacy_applicant.var:sltm_merc_previous_captain
                    var:sltm_merc_claimant ?= scope:sltm_legacy_applicant
                }
                var:sltm_merc_candidate_company.mercenary_company_leader ?= var:sltm_merc_previous_captain
                sltm_legacy_claim_uncontested_trigger = yes
            }
            # A matched company already exists in the world and may depend on this quota.
            set_variable = { name = sltm_merc_resolution value = 2 }
        }
        else_if = {
            limit = {
                NOT = { has_variable = sltm_merc_candidate }
                NOT = { has_variable = sltm_merc_candidate_company }
                NOT = { has_variable = sltm_merc_previous_captain }
                exists = var:sltm_merc_origin
                has_variable = sltm_merc_before_count
                has_variable = sltm_merc_units_reserved
                var:sltm_merc_units_reserved > 0
            }
            sltm_merc_release_unused_reservation_effect = yes
        }
        if = {
            limit = {
                var:sltm_merc_resolution = 4
                OR = { is_alive = no NOT = { var:sltm_merc_sponsor ?= { is_alive = yes } } }
            }
            set_variable = { name = sltm_merc_resolution value = 2 }
        }
        if = {
            limit = { var:sltm_merc_resolution < 4 }
            sltm_merc_clear_request_effect = yes
        }
    }
}

sltm_merc_release_unused_reservation_effect = {
    # Count other applicants' reservations before touching the shared culture ledger.
    set_local_variable = { name = sltm_legacy_required value = var:sltm_merc_units_reserved }
    every_in_global_list = {
        variable = sltm_pending_charters
        limit = {
            is_alive = yes
            NOT = { this = scope:sltm_legacy_applicant }
            var:sltm_merc_origin ?= scope:sltm_legacy_applicant.var:sltm_merc_origin
            has_variable = sltm_merc_units_reserved
        }
        change_local_variable = { name = sltm_legacy_required add = var:sltm_merc_units_reserved }
    }
    if = {
        limit = {
            var:sltm_merc_origin ?= {
                has_variable = sltm_chartered_units
                var:sltm_chartered_units >= local_var:sltm_legacy_required
            }
        }
        if = {
            limit = {
                any_mercenary_company = {
                    mercenary_company_leader ?= {
                        culture ?= scope:sltm_legacy_applicant.var:sltm_merc_origin
                        any_held_title = {
                            is_mercenary_company = yes
                            save_temporary_scope_as = sltm_legacy_band
                            scope:sltm_legacy_applicant = {
                                NOT = { is_target_in_variable_list = { name = sltm_merc_before_titles target = scope:sltm_legacy_band } }
                            }
                        }
                    }
                }
            }
            set_variable = { name = sltm_merc_resolution value = 2 }
        }
        else = {
            var:sltm_merc_origin = {
                change_variable = { name = sltm_chartered_units subtract = scope:sltm_legacy_applicant.var:sltm_merc_units_reserved }
                sltm_merc_capacity_effect = yes
                if = {
                    limit = { has_variable = sltm_charter_capacity_failed }

                }
                else = { scope:sltm_legacy_applicant = { set_variable = { name = sltm_merc_resolution value = 1 } } }
            }
        }
    }
}
