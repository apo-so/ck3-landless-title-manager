# Character scope. Only the current native realm-law keys are addressed; other mods' laws are never cleared.
sltm_remember_realm_law_effect = {
    if = {
        limit = { has_realm_law = $LAW$ }
        add_to_variable_list = { name = sltm_identity_laws target = flag:$LAW$ }
    }
}
sltm_reinstate_realm_law_effect = {
    if = {
        limit = {
            is_target_in_variable_list = { name = sltm_identity_laws target = flag:$LAW$ }
            NOT = { has_realm_law = $LAW$ }
        }
        add_realm_law_skip_effects = $LAW$
    }
}
sltm_discard_transient_law_effect = {
    if = {
        limit = {
            has_realm_law = $LAW$
            NOT = { is_target_in_variable_list = { name = sltm_identity_laws target = flag:$LAW$ } }
        }
        remove_realm_law = $LAW$
    }
}
sltm_verify_realm_law_effect = {
    if = {
        limit = {
            is_target_in_variable_list = { name = sltm_identity_laws target = flag:$LAW$ }
            NOT = { has_realm_law = $LAW$ }
        }
        scope:sltm_actor = { set_variable = sltm_travel_identity_failed }
    }
}
