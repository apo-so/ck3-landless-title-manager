# Same province contract for creation and editing: barony itself; county capital; higher title capital county.
sltm_pick_province_effect = {
 remove_variable = $DEST$
 clear_saved_scope = sltm_input_province
 if = { limit = { exists = $SOURCE$ }
  $SOURCE$ = {
   if = { limit = { tier = tier_barony exists = title_province } title_province = { save_scope_as = sltm_input_province } }
   else_if = { limit = { tier = tier_county exists = title_province } title_province = { save_scope_as = sltm_input_province } }
   else_if = { limit = { exists = title_capital_county.title_province } title_capital_county.title_province = { save_scope_as = sltm_input_province } }
  }
 }
 if = { limit = { exists = scope:sltm_input_province } set_variable = { name = $DEST$ value = scope:sltm_input_province } }
}

# CHARACTER actor scope. Keep independent counts: an invalid reference may be skipped by an iterator.
sltm_create_capture_before_effect = {
 clear_variable_list = sltm_before_titles
 clear_variable_list = sltm_before_domiciles
 set_variable = { name = sltm_before_titles_count value = 0 }
 set_variable = { name = sltm_before_domiciles_count value = 0 }
 scope:sltm_holder = {
  every_held_title = {
   set_variable = { name = sltm_pre_create_holder value = holder }
   remove_variable = sltm_pre_create_domicile
   set_variable = { name = sltm_pre_had_domicile value = 0 }
   if = { limit = { exists = title_domicile }
    set_variable = { name = sltm_pre_create_domicile value = title_domicile }
    set_variable = { name = sltm_pre_had_domicile value = 1 }
   }
   save_scope_as = sltm_old_title
   scope:sltm_actor = {
    add_to_variable_list = { name = sltm_before_titles target = scope:sltm_old_title }
    change_variable = { name = sltm_before_titles_count add = 1 }
   }
   if = { limit = { exists = title_domicile }
    title_domicile = { save_scope_as = sltm_old_domicile }
    scope:sltm_actor = {
     if = { limit = { NOT = { any_in_list = { variable = sltm_before_domiciles this = scope:sltm_old_domicile } } }
      add_to_variable_list = { name = sltm_before_domiciles target = scope:sltm_old_domicile }
      change_variable = { name = sltm_before_domiciles_count add = 1 }
      scope:sltm_old_domicile = {
       remove_variable = sltm_pre_owner
       remove_variable = sltm_pre_location
       set_variable = { name = sltm_pre_had_owner value = 0 }
       set_variable = { name = sltm_pre_had_location value = 0 }
       set_variable = { name = sltm_pre_main value = 0 }
       if = { limit = { exists = owner } set_variable = { name = sltm_pre_owner value = owner } set_variable = { name = sltm_pre_had_owner value = 1 } }
       if = { limit = { exists = domicile_location } set_variable = { name = sltm_pre_location value = domicile_location } set_variable = { name = sltm_pre_had_location value = 1 } }
       if = { limit = { has_domicile_building = sltm_residence_main_01 } set_variable = { name = sltm_pre_main value = 1 } }
      }
     }
    }
   }
  }
 }
 remove_variable = sltm_before_default
 remove_variable = sltm_before_primary
 remove_variable = sltm_before_liege
 if = { limit = { exists = scope:sltm_holder.domicile } set_variable = { name = sltm_before_default value = scope:sltm_holder.domicile } }
 if = { limit = { exists = scope:sltm_holder.primary_title } set_variable = { name = sltm_before_primary value = scope:sltm_holder.primary_title } }
 if = { limit = { exists = scope:sltm_holder.liege } set_variable = { name = sltm_before_liege value = scope:sltm_holder.liege } }
 sltm_read_government_effect = { PERSON = scope:sltm_holder DEST = sltm_before_government }
}

sltm_create_capture_after_effect = {
 remove_variable = sltm_after_default
 remove_variable = sltm_after_primary
 remove_variable = sltm_after_liege
 if = { limit = { exists = scope:sltm_holder.domicile } set_variable = { name = sltm_after_default value = scope:sltm_holder.domicile } }
 if = { limit = { exists = scope:sltm_holder.primary_title } set_variable = { name = sltm_after_primary value = scope:sltm_holder.primary_title } }
 if = { limit = { exists = scope:sltm_holder.liege } set_variable = { name = sltm_after_liege value = scope:sltm_holder.liege } }
 sltm_read_government_effect = { PERSON = scope:sltm_holder DEST = sltm_after_government }
 set_variable = { name = sltm_after_old_titles_count value = 0 }
 set_variable = { name = sltm_after_old_domiciles_count value = 0 }
 set_variable = { name = sltm_old_titles_preserved value = 1 }
 set_variable = { name = sltm_old_domiciles_preserved value = 1 }
 every_in_list = {
  variable = sltm_before_titles
  scope:sltm_actor = { change_variable = { name = sltm_after_old_titles_count add = 1 } }
  if = { limit = { OR = {
   NOT = { holder = var:sltm_pre_create_holder }
   AND = { var:sltm_pre_had_domicile = 1 OR = { NOT = { exists = var:sltm_pre_create_domicile } NOT = { title_domicile = var:sltm_pre_create_domicile } } }
   AND = { var:sltm_pre_had_domicile = 0 exists = title_domicile }
  } } scope:sltm_actor = { set_variable = { name = sltm_old_titles_preserved value = 0 } } }
  remove_variable = sltm_pre_create_holder
  remove_variable = sltm_pre_create_domicile
  remove_variable = sltm_pre_had_domicile
 }
 every_in_list = {
  variable = sltm_before_domiciles
  scope:sltm_actor = { change_variable = { name = sltm_after_old_domiciles_count add = 1 } }
  if = { limit = { OR = {
   AND = { var:sltm_pre_had_owner = 1 OR = { NOT = { exists = var:sltm_pre_owner } NOT = { owner = var:sltm_pre_owner } } }
   AND = { var:sltm_pre_had_owner = 0 exists = owner }
   AND = { var:sltm_pre_had_location = 1 OR = { NOT = { exists = var:sltm_pre_location } NOT = { domicile_location = var:sltm_pre_location } } }
   AND = { var:sltm_pre_had_location = 0 exists = domicile_location }
   AND = { var:sltm_pre_main = 1 NOT = { has_domicile_building = sltm_residence_main_01 } }
  } } scope:sltm_actor = { set_variable = { name = sltm_old_domiciles_preserved value = 0 } } }
  remove_variable = sltm_pre_owner
  remove_variable = sltm_pre_location
  remove_variable = sltm_pre_had_owner
  remove_variable = sltm_pre_had_location
  remove_variable = sltm_pre_main
 }
 if = { limit = { NOT = { var:sltm_after_old_titles_count = var:sltm_before_titles_count } } set_variable = { name = sltm_old_titles_preserved value = 0 } }
 if = { limit = { NOT = { var:sltm_after_old_domiciles_count = var:sltm_before_domiciles_count } } set_variable = { name = sltm_old_domiciles_preserved value = 0 } }
}
