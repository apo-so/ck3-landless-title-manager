sltm_ui_open_effect = {
 sltm_windows_capture_origin_effect = yes
 sltm_ui_open_fixed_effect = yes
}
sltm_ui_open_fixed_effect = {
 if = { limit = { exists = var:sltm_lvm_selected_title }
  sltm_edit_initialize_effect = yes
  set_variable = { name = sltm_ui_origin value = yes }
  sltm_ui_resume_effect = yes
  remove_variable = sltm_lvm_open
 }
}
sltm_ui_resume_effect = {
 if = { limit = { has_variable = sltm_edit_pending exists = var:sltm_edit_target has_variable = sltm_ui_origin }
  set_variable = { name = sltm_ui_revision value = var:sltm_workflow_revision }
  set_variable = { name = sltm_ui_open value = yes }
  sltm_ui_sync_effect = yes
 }
}
sltm_ui_reopen_effect = {
 if = { limit = { exists = var:sltm_edit_target }
  set_variable = { name = sltm_lvm_selected_title value = var:sltm_edit_target }
  sltm_ui_open_fixed_effect = yes
 }
}
sltm_ui_prepare_event_effect = {
 # Capture current revision once for the new event chain, then hide native window.
 save_scope_value_as = { name = sltm_workflow_session value = var:sltm_workflow_revision }
 remove_variable = sltm_ui_open
}
sltm_ui_close_effect = {
 sltm_workflow_new_session_effect = yes
 sltm_edit_clear_draft_effect = yes
}
