sltm_lvm_select_title_gui = {
    scope = landed_title
    saved_scopes = {
        player
    }

    effect = {
        save_scope_as = sltm_lvm_clicked_title
        scope:player = {
            set_variable = {
                name = sltm_lvm_selected_title
                value = scope:sltm_lvm_clicked_title
            }
            # 2026-07-18（V3.3-D）切换选中头衔即解除剥夺/销毁的“确认”待触发状态
            remove_variable = sltm_lvm_arm_revoke
            remove_variable = sltm_lvm_arm_destroy
            remove_variable = sltm_lvm_selected_character
            clear_variable_list = sltm_lvm_selected_holder_titles
            set_variable = { name = sltm_lvm_selected_holder_title_count value = 0 }
            # 2026-07-18（V3.3-F）重建选中头衔持有者的“全部头衔”列表
            sltm_lvm_rebuild_selected_title_holder_all_titles_effect = yes
        }
    }
}
