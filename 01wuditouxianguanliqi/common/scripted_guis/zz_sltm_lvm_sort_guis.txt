# 无地头衔管理器：排序 scripted GUI（V2.6 批次 7）
# 头衔页排序只设纯头衔维度（status/type/tier/role），绝不读取人物属性。
# 每个排序 gui：设对应的 sltm_lvm_title/hlder_sort 旗标 → 只重算对应页。
# 方向切换 gui：在 yes/no 之间翻转 sltm_lvm_*_sort_desc 旗标。

# ---------- 头衔页排序字段 ----------
sltm_lvm_sort_titles_status_gui = {
    scope = character
    effect = {
        set_variable = { name = sltm_lvm_title_sort value = flag:status }
        sltm_lvm_recompute_title_view_effect = yes
    }
}

# ---------- 排序方向切换 ----------
sltm_lvm_toggle_title_sort_order_gui = {
    scope = character
    effect = {
        if = { limit = { var:sltm_lvm_title_sort_desc = flag:yes } set_variable = { name = sltm_lvm_title_sort_desc value = flag:no } }
        else = { set_variable = { name = sltm_lvm_title_sort_desc value = flag:yes } }
        sltm_lvm_recompute_title_view_effect = yes
    }
}
sltm_lvm_toggle_holder_sort_order_gui = {
    scope = character
    effect = {
        if = { limit = { var:sltm_lvm_holder_sort_desc = flag:yes } set_variable = { name = sltm_lvm_holder_sort_desc value = flag:no } }
        else = { set_variable = { name = sltm_lvm_holder_sort_desc value = flag:yes } }
        sltm_lvm_recompute_holder_view_effect = yes
    }
}
