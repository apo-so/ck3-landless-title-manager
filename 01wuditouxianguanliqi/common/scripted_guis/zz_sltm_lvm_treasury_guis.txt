# R10-35A Fix1 — 国库管理全封臣筛选、选择与分页 GUI

sltm_lvm_select_treasury_character_gui = {
    scope = character
    saved_scopes = { player }
    effect = {
        save_scope_as = sltm_lvm_clicked_treasury_character
        scope:player = { set_variable = { name = sltm_lvm_treasury_selected_character value = scope:sltm_lvm_clicked_treasury_character } }
        # R8-B2P3F1：独立保存预算 Probe 目标，不受管理器选择校验/重开清理影响。
        scope:player = { set_variable = { name = sltm_lvm_budget_probe_target value = scope:sltm_lvm_clicked_treasury_character } }
        # 恢复逐团兵士费用明细列表；不恢复任何当前实际总额数值回写/聚合测试。
        sltm_lvm_treasury_rebuild_maa_expense_probe_effect = yes
    }
}
