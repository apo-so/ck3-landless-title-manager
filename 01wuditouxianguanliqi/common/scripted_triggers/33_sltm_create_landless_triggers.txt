# 创立无地头衔 · 两步选择窗口的触发器
# 只读草稿变量，不写任何 live 流程变量。

# 窗口是否打开
sltm_clw_open_trigger = {
	has_variable = sltm_clw_open
 has_variable = sltm_create_route_clw
 has_variable = sltm_create_pending
 var:sltm_clw_revision = var:sltm_workflow_revision
}

# 第 1 步（未设置时按第 1 步处理，避免存档里缺变量把两页都藏掉）
sltm_clw_step_one_trigger = {
	OR = {
		NOT = { has_variable = sltm_clw_step }
		var:sltm_clw_step = 1
	}
}

# 第 2 步
sltm_clw_step_two_trigger = {
	has_variable = sltm_clw_step
	var:sltm_clw_step = 2
}

# 已选类别
sltm_clw_has_category_trigger = {
	has_variable = sltm_clw_kind
}

# 已选类别是「住所头衔」
sltm_clw_category_is_domicile_trigger = {
	has_variable = sltm_clw_kind
	var:sltm_clw_kind ?= 3
}

# 已选住所类型
sltm_clw_has_domicile_type_trigger = {
	has_variable = sltm_clw_domicile_type
 OR = { var:sltm_clw_domicile_type = flag:camp var:sltm_clw_domicile_type = flag:estate var:sltm_clw_domicile_type = flag:yurt var:sltm_clw_domicile_type = flag:east_asian_estate var:sltm_clw_domicile_type = flag:ecclesiastical_domicile var:sltm_clw_domicile_type = flag:japanese_manor var:sltm_clw_domicile_type = flag:sltm_residence var:sltm_clw_domicile_type = flag:merchant_company }
}

# 第 2 步但尚未选住所
sltm_clw_domicile_missing_trigger = {
	sltm_clw_step_two_trigger = yes
	NOT = { has_variable = sltm_clw_domicile_type }
}

# 主按钮此刻可以执行：第 1 步需有类别；第 2 步需有住所
sltm_clw_supported_trigger = {
 has_variable = sltm_clw_kind
 OR = { var:sltm_clw_kind ?= 1 var:sltm_clw_kind ?= 2 var:sltm_clw_kind ?= 3 var:sltm_clw_kind ?= 4 var:sltm_clw_kind ?= 5 var:sltm_clw_kind ?= 6 var:sltm_clw_kind ?= 7 var:sltm_clw_kind ?= 9 var:sltm_clw_kind ?= 10 var:sltm_clw_kind ?= 11 var:sltm_clw_kind ?= 12 }
}
sltm_clw_primary_ready_trigger = {
 var:sltm_create_holder ?= { is_alive = yes }
 sltm_clw_supported_trigger = yes
	OR = {
		AND = { sltm_clw_step_one_trigger = yes sltm_clw_has_category_trigger = yes }
		AND = { sltm_clw_step_two_trigger = yes sltm_clw_category_is_domicile_trigger = yes sltm_clw_has_domicile_type_trigger = yes }
	}
}

# 第 1 步且选了住所头衔：主按钮变成「下一步」
sltm_clw_primary_is_next_trigger = {
	sltm_clw_step_one_trigger = yes
	sltm_clw_category_is_domicile_trigger = yes
}

sltm_atlas_read_trigger = {
   OR = {
     AND = { has_variable = sltm_atlas_view var:sltm_atlas_view = 2 }
     AND = { NOT = { has_variable = sltm_atlas_view } sltm_clw_has_category_trigger = yes }
   }
 }

sltm_atlas_can_create_trigger = {
 NOT = { AND = { var:sltm_clw_kind ?= 3 var:sltm_clw_domicile_type ?= flag:sltm_residence } }
 NOT = { AND = { var:sltm_clw_kind ?= 2 var:sltm_create_tier = 1 } }
 sltm_clw_open_trigger = yes
 sltm_clw_primary_ready_trigger = yes
 NOT = { sltm_clw_primary_is_next_trigger = yes }
 NOT = { has_variable = sltm_atlas_picker }
}

sltm_atlas_institution_trigger = {
    has_variable = sltm_clw_kind
    OR = { var:sltm_clw_kind ?= 1 var:sltm_clw_kind ?= 6 var:sltm_clw_kind ?= 7 var:sltm_clw_kind ?= 8 var:sltm_clw_kind ?= 9 var:sltm_clw_kind ?= 10 var:sltm_clw_kind ?= 11 var:sltm_clw_kind ?= 12 }
}
# Actor scope; evaluated before the commit clears its saved identity scopes.
sltm_order_government_matches_trigger = {
    exists = scope:sltm_order_expected_government
    scope:sltm_holder ?= { government_type ?= scope:sltm_order_expected_government }
}

sltm_special_output_trigger = {
    exists = scope:sltm_created
    OR = {
        AND = { var:sltm_create_special_kind = 1 scope:sltm_created ?= { has_clerical_region = yes any_county_in_clerical_region = { always = yes } } }
        AND = {
            OR = { var:sltm_create_special_kind = 6 var:sltm_create_special_kind = 7 }
            scope:sltm_created ?= { is_holy_order = yes }
            scope:sltm_new_order.title ?= scope:sltm_created
            scope:sltm_new_order.leader ?= scope:sltm_holder
            sltm_order_government_matches_trigger = yes
            scope:sltm_order_base.lessee_title ?= scope:sltm_created
            OR = {
                AND = { var:sltm_create_special_kind = 6 scope:sltm_new_order ?= { is_military_order = yes } }
                AND = { var:sltm_create_special_kind = 7 scope:sltm_new_order ?= { is_military_order = no } }
            }
        }
        AND = { var:sltm_create_special_kind = 9 scope:sltm_holder.faith.religious_head_title ?= scope:sltm_created scope:sltm_holder.faith.religious_head ?= scope:sltm_holder }
        AND = { var:sltm_create_special_kind = 12 scope:sltm_created ?= { holder = scope:sltm_holder sltm_real_merchant_company_trigger = yes } }
        AND = { var:sltm_create_special_kind = 11 scope:sltm_created ?= { is_landless_type_title = yes is_figurehead_title = yes NOT = { exists = title_domicile } } sltm_ceremonial_government_matches_trigger = yes }
        AND = { var:sltm_create_special_kind = 10 scope:sltm_holder.clerical_elector_title ?= scope:sltm_created scope:sltm_holder.faith ?= { any_clerical_elector_title = { this = scope:sltm_created } } }
    }
}

sltm_verified_created_object_trigger = {
    trigger_if = {
        limit = { var:sltm_create_kind = 1 }
        scope:sltm_created ?= { is_titular = yes is_landless_type_title = yes NOT = { exists = title_domicile } }
    }
    trigger_else_if = { limit = { var:sltm_create_kind = 4 } sltm_special_output_trigger = yes }
    trigger_else = { scope:sltm_created ?= { exists = title_domicile title_domicile = { owner = scope:sltm_holder } } sltm_created_domicile_type_matches_trigger = yes }
}

sltm_has_title_rule_trigger = { OR = { has_title_law = $LAW$ has_title_law = sltm_title_$LAW$ } }

sltm_hof_confirmation_current_trigger = {
    var:sltm_hof_confirm_title ?= var:sltm_create_holder.faith.religious_head_title
    var:sltm_hof_confirm_faith ?= var:sltm_create_holder.faith
    var:sltm_hof_confirm_nominee ?= var:sltm_create_holder
    OR = {
     var:sltm_hof_confirm_person ?= var:sltm_create_holder.faith.religious_head
     AND = { NOT = { exists = var:sltm_hof_confirm_person } NOT = { exists = var:sltm_create_holder.faith.religious_head } }
    }
}

# Preferences for automatic selection only. An explicitly marked barony bypasses this filter.
# Existing holy-order leases remain unavailable; secondary mayor/baron holdings keep their transfer path.
# The engine must still return the new order and the exact lessee title before charging.
sltm_order_available_seat_trigger = {
    tier = tier_barony
    exists = holder
    is_under_holy_order_lease = no
    OR = {
        can_be_leased_out = yes
        AND = {
            title_province ?= { is_county_capital = no }
            OR = {
                title_province ?= { has_holding_type = church_holding }
                AND = {
                    holder.primary_title.tier = tier_barony
                    title_province ?= { OR = { has_holding_type = city_holding has_holding_type = castle_holding } }
                }
            }
        }
    }
}

sltm_legacy_request_visible_trigger = {
    var:sltm_create_holder ?= {
        OR = { has_variable = sltm_merc_sponsor is_target_in_global_variable_list = { name = sltm_pending_charters target = this } }
        OR = { var:sltm_merc_sponsor ?= root NOT = { var:sltm_merc_sponsor ?= { is_alive = yes } } }
    }
}

# Rank IDs in the atlas are county=1 through hegemony=5. This warning covers the two selectable creator backends.
sltm_atlas_promotes_primary_trigger = {
    OR = { var:sltm_clw_kind ?= 2 var:sltm_clw_kind ?= 3 var:sltm_clw_kind ?= 4 var:sltm_clw_kind ?= 5 var:sltm_clw_kind ?= 11 }
    exists = var:sltm_create_holder.primary_title
    OR = {
        AND = { var:sltm_create_tier ?= 1 var:sltm_create_holder.primary_title = { tier < tier_county } }
        AND = { var:sltm_create_tier ?= 2 var:sltm_create_holder.primary_title = { tier < tier_duchy } }
        AND = { var:sltm_create_tier ?= 3 var:sltm_create_holder.primary_title = { tier < tier_kingdom } }
        AND = { var:sltm_create_tier ?= 4 var:sltm_create_holder.primary_title = { tier < tier_empire } }
        AND = { var:sltm_create_tier ?= 5 var:sltm_create_holder.primary_title = { tier < tier_hegemony } }
    }
}
sltm_atlas_promotion_current_trigger = {
    var:sltm_rank_revision ?= var:sltm_clw_revision
    var:sltm_rank_holder ?= var:sltm_create_holder
    var:sltm_rank_primary ?= var:sltm_create_holder.primary_title
    var:sltm_rank_tier ?= var:sltm_create_tier
    var:sltm_rank_kind ?= var:sltm_clw_kind
}

# I33: compare the exact government captured by the ceremonial constructor.
sltm_ceremonial_government_matches_trigger = {
 OR = {
  NOT = { exists = scope:sltm_ceremonial_original_ruler }
  AND = { exists = scope:sltm_ceremonial_original_government scope:sltm_holder ?= { government_type ?= scope:sltm_ceremonial_original_government } }
 }
}

# Title scope. A company must have a real, correctly bound company domicile.
sltm_real_merchant_company_trigger = {
    NOT = { has_variable = sltm_dynamic_merchant_residence }
    gdp_merchant_company_title_trigger = yes
    exists = holder
    exists = title_domicile
    title_domicile = { is_domicile_type = merchant_company owner = prev.holder }
}

# Character scope: the player viewing the atlas.
sltm_atlas_estate_choice_trigger = {
    var:sltm_clw_kind ?= 3
    OR = {
        var:sltm_clw_domicile_type ?= flag:estate
        var:sltm_clw_domicile_type ?= flag:east_asian_estate
        var:sltm_clw_domicile_type ?= flag:japanese_manor
    }
}
sltm_atlas_noble_mode_trigger = {
    sltm_atlas_estate_choice_trigger = yes
    has_variable = sltm_family_institution
}
sltm_atlas_patrician_mode_trigger = {
    var:sltm_clw_kind ?= 12
    has_variable = sltm_family_institution
}
sltm_atlas_family_choice_trigger = {
    OR = { var:sltm_clw_kind ?= 12 sltm_atlas_noble_mode_trigger = yes }
}

# Title scope. Compare the native house title, bound domicile and personal holder.
sltm_bound_house_domicile_trigger = {
    is_noble_family_title = yes
    exists = holder
    exists = title_domicile
    title_domicile ?= {
        is_domicile_type = $HOME_TYPE$
        owner = prev.holder
    }
}
sltm_atlas_matching_family_title_trigger = {
    OR = {
        AND = { root.var:sltm_clw_kind ?= 12 sltm_real_merchant_company_trigger = yes }
        AND = { root.var:sltm_clw_kind ?= 3 root.var:sltm_clw_domicile_type ?= flag:estate sltm_bound_house_domicile_trigger = { HOME_TYPE = estate } }
        AND = { root.var:sltm_clw_kind ?= 3 root.var:sltm_clw_domicile_type ?= flag:east_asian_estate sltm_bound_house_domicile_trigger = { HOME_TYPE = east_asian_estate } }
        AND = { root.var:sltm_clw_kind ?= 3 root.var:sltm_clw_domicile_type ?= flag:japanese_manor sltm_bound_house_domicile_trigger = { HOME_TYPE = japanese_manor } }
    }
}
sltm_atlas_owns_family_title_trigger = {
    sltm_atlas_family_choice_trigger = yes
    var:sltm_create_holder ?= { any_held_title = { sltm_atlas_matching_family_title_trigger = yes } }
    exists = var:sltm_create_holder
}
sltm_atlas_family_branch_trigger = {
    sltm_atlas_family_choice_trigger = yes
    NOT = { sltm_atlas_owns_family_title_trigger = yes }
    exists = var:sltm_create_holder.house.house_head
    exists = var:sltm_create_holder.house.noble_family.holder
    var:sltm_create_holder = {
        is_house_head = no
        house.noble_family = {
            holder = root.var:sltm_create_holder.house.house_head
            OR = {
                sltm_real_merchant_company_trigger = yes
                sltm_bound_house_domicile_trigger = { HOME_TYPE = estate }
                sltm_bound_house_domicile_trigger = { HOME_TYPE = east_asian_estate }
                sltm_bound_house_domicile_trigger = { HOME_TYPE = japanese_manor }
            }
        }
    }
}
sltm_atlas_republic_target_trigger = {
    exists = var:sltm_designee_title.holder
    var:sltm_designee_title.holder = {
        has_government = republic_government
        any_held_title = { tier >= tier_county is_landless_type_title = no }
        NOT = { this = root.var:sltm_create_holder }
    }
}
sltm_atlas_family_ready_trigger = {
    sltm_atlas_can_create_trigger = yes
    sltm_atlas_family_choice_trigger = yes
    OR = {
        NOT = { sltm_atlas_patrician_mode_trigger = yes }
        AND = {
            sltm_atlas_republic_target_trigger = yes
            exists = var:sltm_family_republic_title
            var:sltm_family_republic_title = var:sltm_designee_title
            var:sltm_family_republic_ruler = var:sltm_designee_title.holder
        }
    }
}


# Existing membership in the selected realm is a view action, including after reload.
sltm_atlas_current_patrician_trigger = {
    sltm_atlas_patrician_mode_trigger = yes
    sltm_atlas_republic_target_trigger = yes
    exists = var:sltm_family_republic_ruler
    var:sltm_create_holder ?= {
        has_government = republic_government
        liege = root.var:sltm_family_republic_ruler
        exists = house.noble_family
        house.noble_family = { holder = root.var:sltm_create_holder sltm_real_merchant_company_trigger = yes has_title_law = noble_family_succession_law }
    }
}

# Existing companies remain viewable even if a later mod changes a holder's house.
sltm_atlas_company_needs_new_house_trigger = {
    var:sltm_clw_kind ?= 12
    exists = var:sltm_create_holder
    NOT = { exists = var:sltm_create_holder.house }
    NOT = { sltm_atlas_owns_family_title_trigger = yes }
}
