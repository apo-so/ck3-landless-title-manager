    scope:sltm_lvm_filter_player = { OR = {
        has_variable = sltm_lvm_hf_age_0_15_selected

sltm_lvm_v36_matches_age_trigger = {
    OR = {
        NOT = { sltm_lvm_v36_has_age_selection_trigger = yes }
        AND = {
            scope:sltm_lvm_filter_player = { has_variable = sltm_lvm_hf_age_0_15_selected }
            age < 16
        }
        AND = {
            scope:sltm_lvm_filter_player = { has_variable = sltm_lvm_hf_age_16_29_selected }
            age >= 16
                        age < 30
        }
        AND = {
            scope:sltm_lvm_filter_player = { has_variable = sltm_lvm_hf_age_30_49_selected }
            age >= 30
                        age < 50
        }
        AND = {
            scope:sltm_lvm_filter_player = { has_variable = sltm_lvm_hf_age_50_64_selected }
            age >= 50
                        age < 65
        }
        AND = {
            scope:sltm_lvm_filter_player = { has_variable = sltm_lvm_hf_age_65_plus_selected }
            age >= 65
        }
    }
}
