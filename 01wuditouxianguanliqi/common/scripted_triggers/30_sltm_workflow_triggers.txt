  exists = title_domicile

}

# The caller has saved the applicant as sltm_legacy_applicant.
# A living rival's title or company claim keeps the original request open for review.
sltm_legacy_claim_uncontested_trigger = {
    var:sltm_merc_candidate ?= {
        OR = {
            NOT = { has_variable = sltm_merc_claimant }
            var:sltm_merc_claimant ?= scope:sltm_legacy_applicant
        }
        OR = {
            NOT = { has_variable = sltm_merc_native_company }
            var:sltm_merc_native_company ?= scope:sltm_legacy_applicant.var:sltm_merc_candidate_company
        }
    }
    NOT = {
        any_in_global_list = {
            variable = sltm_pending_charters
            is_alive = yes
            NOT = { this = scope:sltm_legacy_applicant }
            OR = {
                var:sltm_merc_candidate ?= scope:sltm_legacy_applicant.var:sltm_merc_candidate
                var:sltm_merc_candidate_company ?= scope:sltm_legacy_applicant.var:sltm_merc_candidate_company
            }
        }
    }
}
